1. Sensor de temperatura con pantalla LCD

Este montaje utiliza un sensor analógico conectado al pin A0, una pantalla LCD de 16×2 y un LED RGB. El programa convierte la lectura analógica en temperatura, la muestra en el puerto serial y la escribe en la pantalla LCD.

¿Qué hace el código?

  1. Incluye la biblioteca LiquidCrystal para controlar la pantalla.
  2. Configura los pines de la LCD: RS en 12, Enable en 13 y datos en 5, 4, 3 y 2.
  3. Lee el sensor de temperatura en A0.
  4. Convierte la lectura analógica mediante la fórmula incluida en el documento.
  5. Envía la temperatura al Monitor Serial y muestra el valor en la LCD.
  6. Cambia el color del LED RGB según la temperatura: azul por debajo de 30 °C, verde entre 30 y 35 °C y rojo por encima de 35 °C.
ElementoConexión o función
Pantalla LCDRS 12, Enable 13, D4 5, D5 4, D6 3, D7 2.
SensorEntrada analógica A0.
LED RGBRojo 9, verde 10, azul 11.
Contraste LCDPin 6 mediante PWM, valor de contraste 1.
Código explicado · temperatura y LCD
// Librería y conexión de la pantalla LCD
#include <LiquidCrystal.h>
LiquidCrystal lcd(12, 13, 5, 4, 3, 2);

int contrastPin = 6;
int contrastValue = 1;
int sensorPin = A0;
int redLed = 9;
int greenLed = 10;
int blueLed = 11;

void setup() {
  Serial.begin(9600);
  pinMode(contrastPin, OUTPUT);
  lcd.begin(16, 2);
  lcd.print("Temperatura: ");
  pinMode(redLed, OUTPUT);
  pinMode(greenLed, OUTPUT);
  pinMode(blueLed, OUTPUT);
}

void loop() {
  int lectura = analogRead(sensorPin);
  float sensorTemp = ((5.0 / 1024) * lectura * 100) - 50;
  int temp = (int)sensorTemp;
  Serial.println(temp);

  if (temp > 35) {
    digitalWrite(redLed, HIGH);
    digitalWrite(greenLed, LOW);
    digitalWrite(blueLed, LOW);
  } else if (temp >= 30 && temp <= 35) {
    digitalWrite(greenLed, HIGH);
    digitalWrite(redLed, LOW);
    digitalWrite(blueLed, LOW);
  } else {
    digitalWrite(blueLed, HIGH);
    digitalWrite(redLed, LOW);
    digitalWrite(greenLed, LOW);
  }

  analogWrite(contrastPin, contrastValue);
  lcd.setCursor(6, 1);
  lcd.print(temp);
  lcd.print("C");
  delay(500);
}

2. Sensor de humo

El segundo montaje usa un sensor de gas/humo en A5, tres LEDs para indicar el nivel de riesgo y un buzzer para emitir una alarma cuando el valor supera los límites definidos.

¿Qué hace el código?

  • Lee el valor del sensor mediante analogRead(A5).
  • Por debajo de 930 muestra una condición fuera de peligro con LED verde.
  • Entre 930 y 960 enciende el LED amarillo y avisa que el peligro se acerca.
  • Por encima de 960 enciende el LED rojo y activa el buzzer.
  • Envía mensajes de estado al Monitor Serial cada 500 milisegundos.
Código explicado · sensor de humo
int humo;
int rojo = 11;
int amarillo = 12;
int verde = 13;
int alarma = 10;

void setup() {
  Serial.begin(9600);
  pinMode(rojo, OUTPUT);
  pinMode(amarillo, OUTPUT);
  pinMode(verde, OUTPUT);
  pinMode(alarma, OUTPUT);
}

void loop() {
  humo = analogRead(A5);

  if (humo < 930) {
    digitalWrite(verde, HIGH);
    digitalWrite(amarillo, LOW);
    digitalWrite(rojo, LOW);
    noTone(alarma);
    Serial.println("ESTAS FUERA DE PELIGRO");
  }

  if (humo >= 930 && humo <= 960) {
    digitalWrite(verde, LOW);
    digitalWrite(amarillo, HIGH);
    digitalWrite(rojo, LOW);
    noTone(alarma);
    Serial.println("OJO! EL PELIGRO SE ACERCA");
  }

  if (humo > 960) {
    digitalWrite(verde, LOW);
    digitalWrite(amarillo, LOW);
    digitalWrite(rojo, HIGH);
    tone(alarma, 440);
    Serial.println("ESTAS EN PELIGRO");
  }
  delay(500);
}

3. Motor de agua

El tercer montaje controla una bomba de 3 V mediante un botón conectado al pin 2. Cada pulsación cambia el estado de la bomba. Un LED verde indica que está encendida y uno rojo indica que está apagada.

¿Qué hace el código?

  • Define el pin 5 para la bomba, el pin 2 para el botón y los pines 8 y 9 para los LEDs.
  • Usa INPUT_PULLUP para el botón.
  • Configura una interrupción que detecta la pulsación del botón.
  • Cambia la variable pumpRunning entre encendido y apagado.
  • Activa el LED verde junto con la bomba y el LED rojo cuando la bomba está detenida.
Código explicado · motor de agua
#define PUMP_PIN 5
#define BUTTON_PIN 2
#define GREEN_LED_PIN 8
#define RED_LED_PIN 9

volatile bool pumpRunning = false;
bool greenLedState = LOW;
bool redLedState = LOW;

void setup() {
  pinMode(PUMP_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  pinMode(GREEN_LED_PIN, OUTPUT);
  pinMode(RED_LED_PIN, OUTPUT);
  attachInterrupt(digitalPinToInterrupt(BUTTON_PIN), buttonPressed, FALLING);
  digitalWrite(PUMP_PIN, LOW);
  digitalWrite(GREEN_LED_PIN, LOW);
  digitalWrite(RED_LED_PIN, LOW);
}

void loop() {
  if (pumpRunning) {
    greenLedState = HIGH;
    redLedState = LOW;
    digitalWrite(PUMP_PIN, HIGH);
  } else {
    greenLedState = LOW;
    redLedState = HIGH;
    digitalWrite(PUMP_PIN, LOW);
  }
  digitalWrite(GREEN_LED_PIN, greenLedState);
  digitalWrite(RED_LED_PIN, redLedState);
}

void buttonPressed() {
  pumpRunning = !pumpRunning;
}
Nota de seguridad: la bomba debe utilizar una alimentación compatible y una etapa de potencia adecuada. No se debe alimentar un motor directamente desde un pin de Arduino si supera la corriente permitida.

Relación con el proyecto contra incendios

Las tres prácticas sirven como base para el proyecto de detección y control de incendios: el sensor de temperatura aporta medición, el sensor de humo aporta detección de riesgo y el motor de agua representa el actuador que puede activar un rociador mediante una interfaz de potencia.
